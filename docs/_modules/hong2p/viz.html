def is_categorical(values: pd.Series) -> bool:
    """Returns whether input seems categorical (vs continuous), based mainly on dtype.

    Values with `int` dtype are considered continuous if they have any negative values,
    otherwise categorical.
    """
    # TODO only count int as categorical if min is 0? or if there are no negative
    # values?
    # TODO maybe also require a sufficiently small number of unique ints?
    # (< ~20-30?)
    #
    # assume we just have continuous/categorical now, and that if not continuous
    # it's categorical
    dtype = values.dtype
    if dtype == float or (dtype == int and values.min() < 0):
        categorical = False
    else:
        categorical = True

    return categorical

@contextlib.contextmanager
def temp_seed(seed):
    """Context manager to temporarily set numpy global random seed.

    Restores previous random state on exit.
    """
    state = np.random.get_state()
    np.random.seed(seed)
    try:
        yield
    finally:
        np.random.set_state(state)

def stripplot(*args, seed: int = 1, **kwargs) -> Axes:
    """Wraps `sns.stripplot` to add `seed=` kwarg for fixing random `jitter=`.
    """
    import seaborn as sns
    with temp_seed(seed):
        return sns.stripplot(*args, **kwargs)

            used, rather than chosen within this function.

            `name2palette` value will all share a common value -> color mapping, and
            thus also share an entry in either `for_legends` or `for_cbars`.
            If `add_legends_and_colorbars` is used to draw the legends and colorbars,
            this will also mean they will share a legend or colorbar.

        var_delim: used to join variable names to make legend/colorbar titles

        na_color: used as NaN color for values using dict palettes (should be distinct
            from colors in dict palettes, though this isn't checked)

    Returns:
        color_df: frame of same shape as `df`, with all values RGB 3-tuple colors.

        for_legends: list of dicts, each with keyword arguments for a call to
            `fig.legend`. Can be passed to `add_legends_and_colorbars` argument of the

def format_index_row(row: pd.Series, *, delim: str = ' / ',

def callable_ticklabels(plot_fn: Callable):
    """Allows [x/y]ticklabel functions evaluated on indices of `df` (`plot_fn(df, ...)`)

    First parameter to `plot_fn` must be a `pandas.DataFrame` (or `xarray.DataArray`
    convertable to one) to compute the `str` ticklabels from. `plot_fn` must also accept
    the kwargs '[x|y]ticklabels', if they will be passed.

    Callable '[x|y]ticklabels' are evaluated on each row of the corresponding index (see
    `matlabels`), and passing `True` is equivalent to passing `format_index_row`.
    'delim' and 'float_format' kwargs are passed to `format_index_row` (and not to
    `plot_fn`).

    If the input to the decorated function is a `xarray.DataArray`, the decorated
    function will receive a `DataFrame` (created via `arr.to_pandas()`) as input
    instead. Note that only 'dimension coordinates' (see xarray terminology page) are
    preserved when converting from `DataArray` to `DataFrame`, so use
    `<arr>.set_index(<dimension name>=...)` appropriately before passing the array to a
    wrapped function.
    """
    def _check_bools(ticklabels, **kwargs):
        """Returns fn calling `format_index_row` if `ticklabels` is True, else input.

        `kwargs` are passed to `format_index_row`.
        """
        if ticklabels == True:
            # TODO TODO probably also default to this in case ticklabels=None
            # (might just need to modify some of the calling code)
            format_fn = lambda x: format_index_row(x, **kwargs)
            return format_fn

        # NOTE: this broke ticklabel=False support sns.clustermap has (from
        # sns.heatmap). currently moved this processing into matshow, which had come to
        # expect this behavior (but maybe revert the matshow changes alongside when this
        # code was added?)
        #elif ticklabels == False:
        #    return None

        # TODO delete after checking code that could be affected to see if it trips.
        #
        # I.e. if it is specifically the callable `str`, which I had sometimes used for
        # single level indices in the past, but have encountered issues with and made
        # `format_index_row` to replace.
        elif ticklabels is str:
            warnings.warn('replace usage of str with hong2p.viz.format_index_row')
            return ticklabels

        else:
            return ticklabels

    # TODO should this also include numpy.ndarray subclasses in input types?
    @functools.wraps(plot_fn)
    def wrapped_plot_fn(df_or_arr: DataFrameOrDataArray, *args, **kwargs):

        # TODO use inspect to get any kwargs to format_index_row, then pass that subset
        # of kwargs thru to _check_bools (so i don't have to keep manually including in
        # this list?)?
        default_format_fn_kwarg_names = ['delim', 'float_format', '_debug']
        _pass_thru_to_plot_fn = ['_debug']
        default_format_fn_kws = {
            k: kwargs[k] for k in default_format_fn_kwarg_names if k in kwargs
        }
        for k in default_format_fn_kws.keys():
            if k not in _pass_thru_to_plot_fn:
                kwargs.pop(k)
        #

        if isinstance(df_or_arr, xr.DataArray):
            # Requires df_or_arr to be <=2d
            df = df_or_arr.to_pandas()
        else:
            df = df_or_arr

        if 'xticklabels' in kwargs:
            xticklabels = _check_bools(kwargs['xticklabels'], **default_format_fn_kws)
            if callable(xticklabels):
                xticklabels = col_labels(df, xticklabels)

            kwargs['xticklabels'] = xticklabels

        if 'yticklabels' in kwargs:
            yticklabels = _check_bools(kwargs['yticklabels'], **default_format_fn_kws)
            if callable(yticklabels):
                yticklabels = row_labels(df, yticklabels)

            kwargs['yticklabels'] = yticklabels

        return plot_fn(df, *args, **kwargs)

    return wrapped_plot_fn

def is_centered_norm(norm: Union[type, Normalize]) -> bool:
    """Returns whether `Normalize` subclass (or instance) is centered on some value.

    True for `TwoSlopeNorm`, `CenteredNorm`, and `SymLogNorm` (implicitly centered on
    0), as well as any others with a `vcenter` attribute.
    """
    # hasattr checks seem to (currently) work for both the uninitialized classes
    # (type(x) is type) as well as instances (isinstance(x, Normalize), for any
    # parameters in each specific norm constructor
    if hasattr(norm, 'vcenter'):
        return True

    # matplotlib < 3.5 (at least 3.4.2) only defines vcenter on instances, so hasattr
    # check above fails for these classes
    if isinstance(norm, type) and issubclass(norm, (TwoSlopeNorm, CenteredNorm)):
        return True

    # NOTE: SymLogNorm is assumed to be centered about 0, and does not have a vcenter
    # parameter
    if ((isinstance(norm, type) and issubclass(norm, SymLogNorm)) or
        isinstance(norm, SymLogNorm)):

        return True

    return False

def get_mpl_norm(norm: Union[str, type] = Normalize, data: Optional[ArrayLike] = None,
    *, vmin: Optional[float] = None, vmax: Optional[float] = None,
    vcenter: Optional[float] = None, _debug: bool = False, **norm_kws) -> Normalize:
    """Returns instantiated `Normalize` subclass, with limits from args / data.

    Args:
        norm: uninstantiated `Normalize` subclass, or 'two-slope' (`TwoSlopeNorm`) /
            'centered' (`CenteredNorm`)

        data: if passed, used to set any of `vmin`/`vmax` not passed

        vmin: required (directly or via `data`), unless `halfrange` passed in
            `norm_kws` (only allowed for `CenteredNorm`)

        vmax: see `vmin`

        vcenter: for centered norms (see `is_centered_norm`, though not supported for
            `SymLogNorm`). defaults to 0. if `vmin > vcenter`, `vmin` is set to 0 (with
            a warning), and if `vmin == vcenter`, `vmin` is decreased by
            `EPSILON_FROM_VCENTER`.

        **norm_kws: passed to `norm` constructor

    NOTE: `TwoSlopeNorm` does not take a `clip` kwarg. In recent matplotlib (tested
    with 3.11), it maps values beyond `vmin`/`vmax` to -inf/inf (so the colormap's
    under/over colors are used), but this may differ in older versions.
    """
    if norm == 'two-slope':
        norm = TwoSlopeNorm
    elif norm == 'centered':
        norm = CenteredNorm
    # TODO support LogNorm and SymLogNorm too (via strs)?
    # TODO rename all str norm options to be spelled exactly as classes?

    assert not isinstance(norm, Normalize), ('pass the class of the Normalize object '
        'instead of an an already initialized object of that class'
    )

    # TODO did any plot_fns actually take norm=<str> directly (and would have to be
    # other than the two above)? old code seemed like it would pass str other than
    # those explicitly listed in conditionals (in what is now get_mpl_norm) to norm=
    # kwarg here. are there any that i might want to use that do take str norm?
    assert not isinstance(norm, str), (f'want to support this str {norm=} (pass thru to'
        ' plot_fn directly, like old code did [intentionally or not]?'
    )

    # TODO or issubclass(norm, Norm) instead?
    assert isinstance(norm, type) and issubclass(norm, Normalize), (
        'expected norm to be one of the str listed above, or an uninstantiated '
        f'subclass of mpl.colors.Normalize. got {norm=}'
    )

    dmin = None
    dmax = None
    halfrange = norm_kws.pop('halfrange', None)
    if halfrange is not None:
        # TODO do any other Normalize subclasses take halfrange?
        assert issubclass(norm, CenteredNorm), (f'only expected to receive {halfrange=}'
            f' for CenteredNorm. {norm=} {hasattr(norm, "halfrange")=}'
        )
        assert vmin is None, f'{vmin=} should not be specified if halfrange is'
        assert vmax is None, f'{vmax=} should not be specified if halfrange is'

        # TODO also assert data is None (meh), or leave that one, and err (as doing now)
        # for explicit vmin/vmax?
    else:
        if data is not None:
            dmin = data.min().min()
            dmax = data.max().max()
            if vmin is None:
                vmin = dmin
            if vmax is None:
                vmax = dmax

        assert vmin is not None, ('vmin=<float> required for all Normalize objects '
            "(except if norm=CenteredNorm/'centered' and halfrange passed). must either"
            ' pass data or vmin'
        )
        assert vmax is not None, ('vmax=<float> required for all Normalize objects '
            "(except if norm=CenteredNorm/'centered' and halfrange passed). must either"
            ' pass data or vmax'
        )
        # TODO should i assert vmax > vmin? (equality?)
        assert vmax >= vmin, f'{vmax=} < {vmin=}'

        if vmax == vmin:
            warnings.warn(f'{vmin=} == {vmax=}. mistake?')

    centered_norm = is_centered_norm(norm)

    # SymLogNorm is a "centered" norm, but it's implicitly around 0, and does not take
    # vcenter
    if not centered_norm or issubclass(norm, SymLogNorm):
        assert vcenter is None, 'explicit vcenter not supported in this case'

    kws = dict()
    if centered_norm:
        if vcenter is None:
            vcenter = 0.0

        if halfrange is None:
            # prob want these checks even for SymLogNorm, where vcenter isn't passed,
            # but is implied to be 0
            assert vcenter < vmax, f'{vmax=} <= {vcenter=}'

            # TODO delete (prob?)? or delete code below center vmin to 0 -
            # EPSILON_FROM_VCENTER if it starts > vcenter? allowing equality for vmin,
            # to be consistent w/ handling below
            #assert vmin <= vcenter, f'{vmin=} > {vcenter=}'
            #

            # TODO still do this? (would have to comment the assertion above that
            # vmin <= vcenter
            if vmin > vcenter:
                warnings.warn(f'setting vmin=0 since {vmin=} > {vcenter=}')
                vmin = 0

            # NOTE: if (vmin < vcenter < vmax) is not strictly True
            # (e.g. if vmin == vcenter), initializing TwoSlopeNorm would raise:
            # ValueError: vmin, vcenter, and vmax must be in ascending order
            if vmin == vcenter:
                old_vmin = vmin
                vmin = vcenter - EPSILON_FROM_VCENTER
                warnings.warn(f'setting vmin=({vcenter - EPSILON_FROM_VCENTER=}) (from '
                    f'{old_vmin}) to make vmin < {vcenter=} True'
                )

            if issubclass(norm, CenteredNorm):
                # NOTE: this has clip=False by default (UNLIKE two slope, which doesn't
                # support clip kwarg)
                #
                # regarding halfrange "Defaults to the largest absolute difference to
                # vcenter for the values in the dataset."
                # TODO how does it do that? via what imshow does w/ norm input?
                assert vmin < 0, (
                    'assumes vmin < 0 for CenteredNorm, with current calculation'
                )
                assert vmax > 0, (
                    'assumes vmax > 0 for CenteredNorm, with current calculation'
                )
                halfrange = max(-vmin, vmax)
                kws = dict(halfrange=halfrange)

        if not issubclass(norm, SymLogNorm):
            kws.update(dict(vcenter=vcenter))
        # TODO compute some default for SymLogNorm's required linthresh (from a
        # percentage of vmin/vmax), if not set via norm_kws?

    # centered norm does not take vmin/vmax. just vcenter/halfrange/clip
    if not issubclass(norm, CenteredNorm):
        kws.update(dict(vmin=vmin, vmax=vmax))

    if _debug:
        print()
        print(f'{dmin=} {dmax=}')

        print(f'{vmin=} {vmax=}')
        if centered_norm:
            print(f'{vcenter=}')

        print(f'{norm=}')
        print(f'{type(norm)=}')
        print(f'{centered_norm=}')
        if halfrange is not None:
            print(f'{halfrange=}')
        print('both kws and norm_kws below used as args to initialize the Normalize '
            '[sub]class'
        )
        print(f'{kws=}')
        print(f'{norm_kws=}')

    norm = norm(**kws, **norm_kws)
    assert isinstance(norm, Normalize), (f'expected instantiated {type(norm)=} to have '
        'isinstance(norm, Normalize)=True'
    )

    # matplotlib < 3.5 (at least 3.4.2) CenteredNorm does not set vmin/vmax from
    # halfrange on init (only lazily, when first called on data)

def add_norm_options(plot_fn: Callable):
    # TODO prob rewrite fn to use str 'centered' / 'two-slope' instead of (more
    # complicated for user) classes
    """Decorator allowing `plot_fn` to be passed uninstantiated `Normalize` subclasses.

    First argument to `plot_fn` must be the data to be plotted. Adds these keyword
    arguments to the wrapped function:

        norm: matplotlib expects str (e.g. 'linear', 'log', 'symlog', 'logit') or
            an instantiated `Normalize` subclass (or `None`), but passing a normalize
            subclass like this conflicts with non-`None` `vmin`|`vmax`. this decorator
            adds the option to pass a non-instantiated `Normalize` class (or 'centered'
            / 'two-slope'. see `get_mpl_norm`), which will receive `vmin`|`vmax`
            (computed from data, if not passed). this will let plotting functions
            determine the range of data to plot, while still allowing use of norms not
            able to be passed a str.

        vmin|vmax|vcenter|norm_kws: passed to `get_mpl_norm`, if `norm` is passed
            uninstantiated. otherwise, `vmin`/`vmax` are passed to `plot_fn`.

        clip_warn: if True, warns if any data is outside [vmin, vmax]

    `plot_fn` will receive an instantiated `norm` (and not `vmin`/`vmax`) if `norm` was
    passed, otherwise `vmin`/`vmax` (and not `norm`). Warns if a centered norm is used
    with a seemingly non-diverging `cmap`.

    See https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.imshow.html
    for descriptions of `vmin`/`vmax`/`norm` kwargs.
    """
    # TODO possible to inspect whether wrapped fn already has defaults for vmin/vmax,
    # and use those if so (for dff_imshow in al_analysis)?
    @functools.wraps(plot_fn)
    def wrapped_plot_fn(data: ArrayLike, *args, clip_warn: bool = False,
        norm: Optional[Union[str, type, Normalize]] = None,
        vmin: Optional[float] = None, vmax: Optional[float] = None,
        vcenter: Optional[float] = None, norm_kws: Optional[ParamDict] = None,
        _debug: bool = False, **kwargs) -> Any:

        to_init_norm = [vmin, vmax, vcenter, norm_kws]
        if not all(x is None for x in to_init_norm):
            assert not isinstance(norm, Normalize), (f'{norm=} was already an '
                'instantiated Normalize subclass, but one of vmin/vmax/vcenter/norm_kws'
                ' was not None (all used to initialize a Normalize class, and probably '
                'useless with existing Normalize instance)'
            )

        if not (norm is None or isinstance(norm, Normalize)):
            if norm_kws is None:
                norm_kws = dict()
            norm = get_mpl_norm(norm, data=data, vmin=vmin, vmax=vmax, vcenter=vcenter,
                _debug=_debug, **norm_kws
            )
            assert isinstance(norm, Normalize)

        if norm is not None and is_centered_norm(norm):
            # using None should give us consistent default behavior w/ plt.get_cmap
            cmap = kwargs.get('cmap', None)
            if not is_cmap_diverging(cmap):
                warnings.warn('using seemingly non-diverging cmap '
                    f'({get_cmap_name(cmap)}) with {norm=}, where center has special '
                    'meaning. probably a mistake!'
                )

            # TODO also support discrete colormap here (instantiating from just str cmap
            # name)? would be to replace stuff like in
            # al_analysis.plot_n_per_odor_and_glom (see also usage of ListedColormap in
            # map_each_series_to_rgb code. may be simpler?)

        if isinstance(norm, Normalize):
            # even a CenteredNorm object (which is not initialized with vmin/vmax),
            # still has these properties
            if vmin is None:
                vmin = norm.vmin
            if vmax is None:
                vmax = norm.vmax

        # TODO test i can actually silence w/ clip_warn
        # TODO should default value of clip_warn depend on value of clip=<bool> to norm
        # object (or should message?)?
        # TODO threshold fraction before warning?
        # TODO only warn here if we can't show clipping in plot
        # (like w/ default TwoSlopeNorm [right?]) (and/or not configured to)
        clip_msgs = []
        if clip_warn and vmin is not None:
            # .<op>().<op>() should work w/ at least ndarrays and DataFrames
            # (e.g. for min/max/sum ops)
            vmin_n_clipped = (data < vmin).sum().sum()
            vmin_clipped_frac = vmin_n_clipped / data.size
            if vmin_clipped_frac > 0:
                clip_msgs.append(f'{vmin_clipped_frac:.4f} of data ({vmin_n_clipped} '
                    f'points) < {vmin=}'
                )

        if clip_warn and vmax is not None:
            vmax_n_clipped = (data > vmax).sum().sum()
            vmax_clipped_frac = vmax_n_clipped / data.size
            if vmax_clipped_frac > 0:
                clip_msgs.append(f'{vmax_clipped_frac:.4f} of data ({vmax_n_clipped} '
                    f'points) > {vmax=}'
                )

        if len(clip_msgs) > 0:
            assert clip_warn
            # TODO have this be an error by default, allowing decorator (/decorated
            # fn) to be run w/ warning instead, so that it can err if any of matshow
            # data is over/under (but more OK/expected to have imshow data
            # over/under) (just watch for plot_fn.__name__ == 'matshow' in output?)
            warnings.warn(f'{plot_fn.__name__} (add_norm_options wrapper): '
                f"{', '.join(clip_msgs)}\nset clip_warn=False to silence"
            )

        # TODO delete
        # see notes in add_colorbar TwoSlopeNorm section for some hints about plots that
        # might be affected by this (maybe more plots affected by this fn than that
        # one?)
        if _debug:
            print(f'{plot_fn.__name__}: end of add_norm_options: {vmin=}, {vmax=}')
        #

        # not passing vmin/vmax, as matplotlib would raise ValueError since norm also
        # passed
        if norm is not None:
            return plot_fn(data, *args, norm=norm, **kwargs)
        else:

    return wrapped_plot_fn

def _mpl_constrained_layout(enable: bool):
    """
    To make decorators for fns that create Figure(s) and need contrained layout on/off.

    Use `@constrained_layout` or `@no_constrained_layout` rather than this directly.

    NOTE: savefig call will often also have to happen inside the same kind of context
    manager, at least for calls like `seaborn.ClusterGrid.savefig`.
    """
    def wrap_plot_fn(plot_fn):
        """
        Args:
            plot_fn: must make figure from start to finish inside, so changing
                constrained layout setting works
        """
        @functools.wraps(plot_fn)
        def wrapped_plot_fn(*args, **kwargs):
            # NOTE: this only changes the rcParam, which is only used for figures
            # created inside plot_fn. if an `ax` from an existing figure is passed in,
            # that figure's layout is not changed.
            # TODO probably want to be able to override (e.g. to deal with matshow's
            # issue where once i start adding text (for group labels), constrained
            # layout can cut them off)
            with mpl.rc_context({'figure.constrained_layout.use': enable}):
                # TODO maybe monkeypatch any .savefig methods on returned object to warn
                # about need to add context manager around that call too? or just
                # monkeypatch to have constrained layout forced to be consistent in any
                # savefig methods?
                return plot_fn(*args, **kwargs)

        return wrapped_plot_fn

    return wrap_plot_fn

def coarsen(df: pd.DataFrame, downsampling_factor: int, *, random_state: int = 1,
    # TODO remove n_init='auto' if updating sklearn. just to silence FutureWarning
    n_init: Union[str, int] = 'auto', **kwargs) -> pd.DataFrame:
    """Returns copy of `df` w/ rows assigned to clusters of size `downsampling_factor`.

    Rows are clustered with K-means (with `len(df) // downsampling_factor` clusters),
    and then assigned to (equal sized) clusters by optimal assignment to cluster
    centers. The largest K-means cluster gets any remainder rows. Cluster IDs are added
    as a new index level named `COARSEN_CLUSTER_COL`.

    Args:
        df: data to cluster rows of

        downsampling_factor: number of rows per cluster

        random_state: passed to `KMeans`

        n_init: passed to `KMeans`

        **kwargs: passed to `KMeans`
    """
    # current strategy from: https://stats.stackexchange.com/questions/8744
    from sklearn.cluster import KMeans
    from scipy.optimize import linear_sum_assignment
    from scipy.spatial.distance import cdist

    # TODO maybe use numpy dtype check instead, to work w/ np.int types?
    assert type(downsampling_factor) is int, 'mainly trying to check it is not float'

    # TODO or need < len(df)//2. shouldn't matter? add a test case hitting that
    assert 1 < downsampling_factor < len(df) / 2, f'{downsampling_factor=}'

    n_clusters = len(df) // downsampling_factor
    # will need to pick a cluster to get this many extra data points
    # (or otherwise handle the mismatch)
    n_extra = len(df) % downsampling_factor
    assert n_clusters * downsampling_factor + n_extra == len(df)

    # in my current version (1.3.0) I need to add explicit n_init to avoid FutureWarning
    # (about default change from =10 to ='auto' in 1.4) (natmix_data/analysis.py had
    # made the other decision, and set it to =10. maybe want to change that? prob
    # doesn't matter much)
    kmeans = KMeans(n_clusters=n_clusters, random_state=random_state, n_init=n_init,
        **kwargs
    )
    est = kmeans.fit(df)

    centers = est.cluster_centers_
    assert centers.shape[-1] == len(df.columns)
    assert len(centers) == n_clusters

    repeats = np.array([downsampling_factor] * n_clusters)
    if n_extra > 0:
        # the actual cluster IDs: [0, n_clusters-1]
        labels = est.labels_
        assert len(labels.shape) == 1 and len(labels) == len(df)

    repeated_dists = np.repeat(cdist(df, centers), repeats, axis=1)

    assert COARSEN_CLUSTER_COL not in index_df.columns

    new_index = pd.MultiIndex.from_frame(index_df)
    df = df.copy()
    df.index = new_index

        method: default 'average' is same as `sns.clustermap`
        metric: default 'euclidean' is same as `sns.clustermap`

            `sns.clustermap`.

        row_coarsen_factor: if passed, rows are first averaged within clusters of
            approximately this size (see `coarsen_within_groups`), within groups

    if z_score not in valid_preproc_kw_values:

    if standard_scale not in valid_preproc_kw_values:
        raise ValueError(f'standard_scale must be in {valid_preproc_kw_values}')

    if cbar_label is not None:
        if cbar_kws is None:
            cbar_kws = dict()

        cbar_kws['label'] = cbar_label

    let_seaborn_compute_linkages = False
    if z_score is not None or standard_scale is not None:
        assert row_linkage is None and col_linkage is None

        if return_linkages:

            # re-implement this preprocessing before my own linkage computations?
            raise NotImplementedError('can not return linkages while z_score or '
                'standard_scale is True'
            )

        if optimal_ordering:
            warnings.warn('disabling optimal_ordering since z_score or standard_scale')
            optimal_ordering = False

        let_seaborn_compute_linkages = True

        kwargs['z_score'] = z_score
        kwargs['standard_scale'] = standard_scale

    # TODO TODO (done? delete?) do i need row_cluster=True to support coarsening within
    # levels of those before clustering? (and same for columns) (see comments below)
    row_colors = kwargs.pop('row_colors', None)
    if row_coarsen_factor is not None:
        # linkage passed in would have to be computed on something coarsened the same
        # way, or else it wouldn't work
        # TODO TODO (done? delete?) should we assume input is already coarsened here
        # (maybe providing option to return it here?), or recompute? recomputing would
        # be cleaner, but potentially compute heavy w/o caching (and how to even cache?)
        # TODO should i make return type a dict already, so i can support arbitrary
        # return values without having to change as much code? (could remove
        # return_linkages flag?)

            # row_colors values? prob more trouble than worth, even if could do it
            # somehow

                assert not row_colors.isna().any().any()

                colors_per_group = row_colors.groupby(level=row_coarsen_by, sort=False

        else:

        assert 'roi' in df.index.names, f'{df.index.names=}'
        # TODO otherwise, how to tell which to average over? last one?
        nonroi_levels = [c for c in df.index.names if c != 'roi']

        if row_colors is not None:
            assert row_colors.index.equals(df.index.droplevel(COARSEN_CLUSTER_COL))

            row_colors = row_colors.groupby(level=nonroi_levels, sort=False).first()

        # TODO anything better than taking the mean? just re-ordering according to

        df = df.groupby(level=nonroi_levels, sort=False).mean()

    # TODO if z-scoring / standard-scaling requested, calculate before in this case
    # (so it actually affects linkage, as it would w/ seaborn version)
    # (currently just disabling optimal ordering in these cases)
    def _linkage(df):
        from scipy.cluster.hierarchy import linkage
        # TODO way to get this to work w/ some NaNs? worth it?
        return linkage(df.values, optimal_ordering=optimal_ordering, method=method,
            metric=metric
        )

    if not let_seaborn_compute_linkages:
        if row_cluster and row_linkage is None:
            row_linkage = _linkage(df)

    if title is not None:
        if title_kws is None:
            title_kws = dict()
        clustergrid.ax_heatmap.set_title(title, **title_kws)

    if xlabel is not None:
        if xlabel_kws is None:
            xlabel_kws = dict()

    if ylabel is not None:
        ylabel_kws = _ylabel_kwargs(
            ylabel_rotation=ylabel_rotation, ylabel_kws=ylabel_kws
        )
        clustergrid.ax_heatmap.set_ylabel(ylabel, **ylabel_kws)

    if not return_linkages:
        return clustergrid
    else:
        # can use scipy.cluster.hierarchy.leaves_list to get order of input data from
        # these linkages
        return clustergrid, row_linkage, col_linkage

def get_cbar_axis(cbar: Colorbar) -> Axis:
    """

def min_tick_for_centered_cmap(vmin: float, *, increment: float = 0.05,
    round_tick_towards0_under_rtol: float = 0.1, _debug: bool = False) -> float:
    """Finds suitable minimum tick for colorbar using a diverging cmap with `vmin` < 0.

    Returns a multiple of `increment` <= `vmin` (or slightly greater than `vmin`,
    if within `round_tick_towards0_under_rtol`).

    Args:
        vmin: minimum value of colorbar (and of data, if colormap is showing the full
            range of the data). should be negative.

        increment: minimum tick will be a multiple of this

        round_tick_towards0_under_rtol: if vmin * increment < nearest increment, by this
            fraction of increment size, set the min tick to that nearest (closer to 0)
            increment.

        For example:
        ```
        ipdb> vmin
        -0.35374016010182685
        ipdb> increment
        0.05
        ipdb> vmin / increment
        -7.074803202036537
        ipdb> np.round(vmin / increment)
        -7.0
        ipdb> vmin / increment - np.round(vmin / increment)
        -0.07480320203653701
        ipdb> np.isclose(vmin / increment, np.round(vmin / increment), atol=0.1)
        True
        ```
    """
    assert vmin < 0, f'expected {vmin=} to be negative'
    assert increment > 0, f'expected {increment=} to be positive'
    assert round_tick_towards0_under_rtol > 0, (
        f'expected {round_tick_towards0_under_rtol=} to be positive'
    )
    n_increments_float: float = np.abs(vmin) / increment
    n_increments_rounded = np.round(n_increments_float)

    if _debug:
        print()
        print(f'{vmin=}')
        print(f'{increment=}')
        print()
        print(f'{np.abs(vmin)=}')
        print(f'{n_increments_float=}')
        print(f'{n_increments_rounded=}')
        print()

    # TODO ever want to round away from 0 (within some other tolerance),
    # or always want cbar to include all the data?
    if n_increments_rounded > n_increments_float:
        if _debug:
            print('n_increments_rounded > n_increments_float (case 1)')

        n = np.ceil(n_increments_float)

    elif np.isclose(n_increments_float, n_increments_rounded,
            atol=round_tick_towards0_under_rtol
        ):
        if _debug:
            print('n_increments_float within rtol of n_increments_rounded (case 2)')

        n = n_increments_rounded
    else:
        if _debug:
            print('else case')

        n = np.ceil(n_increments_float)

    min_tick = -n * increment
    if _debug:
        print(f'{n=}')
        print(f'{min_tick=}')

    # just so output can always check vmin <= min_tick, without having to worry about
    # numerical issues causing min_tick to be insignificantly (due to numerical issues)
    # > vmin
    # TODO add a flag to disable this? (it did make test case w/ vmin=-.99999 and
    # expected_min_tick=-1.0 fail, before changing default tols here)
    # rtol=1e-6 smaller than default 1e-5. default atol is 1e-8
    if np.isclose(min_tick, vmin, rtol=1e-6):
        if _debug:
            print(f'setting {min_tick=} to {vmin=} b/c they are close')
        min_tick = vmin

    # this even change any numerical things? (yes, and gets test to pass w/ ==, rather
    # than np.isclose)
    # TODO flag to disable this?
    min_tick_rounded = np.round(min_tick, 3)
    if np.isclose(min_tick_rounded, min_tick):
        min_tick = min_tick_rounded

    return min_tick

def contour_bbox(contour):
    """Returns dict with 'xmin', 'xmax', 'ymin', 'ymax' of (n, 2) contour vertices.
    """
    _check_contour_array(contour)

    x = contour[:, 0]
    y = contour[:, 1]
    xmin, ymin = contour.min(axis=0)
    xmax, ymax = contour.max(axis=0)
    # TODO maybe return as NamedTuple or something instead?
    return dict(xmin=xmin, xmax=xmax, ymin=ymin, ymax=ymax)

def contour_center_of_mass(contour):
    """Returns (x, y) center of mass of closed polygon of (n, 2) contour vertices.
    """
    _check_contour_array(contour)

    # Taken from ImportanceOfBeingErnests's answer here:
    # https://stackoverflow.com/questions/48168880
    # calculate center of mass of a closed polygon
    x = contour[:, 0]
    y = contour[:, 1]
    g = (x[:-1] * y[1:] - x[1:] * y[:-1])
    A = 0.5 * g.sum()
    cx = ((x[:-1] + x[1:]) * g).sum()
    cy = ((y[:-1] + y[1:]) * g).sum()
    return 1. / (6 * A) * np.array([cx, cy])

def micrometer_depth_title(ax: Axes, zstep_um: float, z_index: int, *,
    as_ylabel: bool = False, as_overlay: bool = False, fontsize: Optional[float] = 8,
    **kwargs) -> None:
    """Labels `ax` with depth of plane, in micrometers (e.g. '-12 um' for z_index=1).

    Args:
        ax: Axes to label

        zstep_um: distance (in um) between planes

        z_index: index of plane `ax` shows. depth is `-zstep_um * z_index`.

        as_ylabel: if True, use `ax.set_ylabel` instead of `ax.title`, to put this
            information on the left (may want to use if have a series of single column
            plots, rather than a series of rows). adds `rotation=0` and `labelpad=16`
            by default. mutually exclusive with `as_overlay`.

        as_overlay: draws on `ax` instead of labelling above / to the side. in top left
            corner by default (`x`/`y` kwargs, in axes coordinates, can be used to
            change position), in white by default. mutually exclusive with `as_ylabel`.

        fontsize: passed to `ax.set_title` (or `ax.set_ylabel`, if `as_ylabel=True`;
            `ax.text` if `as_overlay=True`)

        **kwargs: passed to `ax.set_title` (or `ax.set_ylabel`, if `as_ylabel=True`;
            `ax.text` if `as_overlay=True`)
    """
    curr_z = -zstep_um * z_index

    if not (as_ylabel or as_overlay):
        set_title = ax.set_title
        extra_default_kws = dict()

    elif as_ylabel:
        set_title = ax.set_ylabel
        # w/o labelpad text overlaps w/ images. labelpad=20 worked w/ fontsize=8.
        # labelpad=16 works w/ fontsize=7 (and max title length from "-70 um")
        extra_default_kws = dict(rotation=0, labelpad=16)

    elif as_overlay:
        # color='w' works w/ black background (e.g. w/ cmap='gray'), but would prefer
        # color='k' (black) w/ white background (e.g. w/ cmap='vlag', or other diverging
        # where center is white). plot_rois will change default according to input image
        # + cmap.
        extra_default_kws = dict(x=0.02, y=0.92, color='w')

        # TODO TODO TODO make scalebar font consistent with this
        # (one way or the other. preferably by changing scalebar stuff to match this)
        def set_title(text, x=None, y=None, **kwargs):
            # these should be specified either by input kwargs or by extra_default_kws
            assert x is not None and y is not None
            ax.text(x, y, text, transform=ax.transAxes, **kwargs)

    # TODO replace w/ `kwargs = {**extra_default_kws, **kwargs}`?
    for k, v in extra_default_kws.items():
        if k not in kwargs:
            kwargs[k] = v

    # TODO check formatting of 0 looks good (or disable that here? currently some
    # code excludes this call on that plane, so not sure of formatting)
    # (it currently looks like "-0uM")
    set_title(f'{curr_z:.0f} $\\mu$m', fontsize=fontsize, **kwargs)

def get_default_overlay_color(ax: Axes) -> str:
    """Returns 'w' or 'k' (white/black), depending on current image colors.

    Aims to provide the color that would be higher contrast with respect to the
    background.
    """
    # NOTE: requires that imshow (/equiv) has been called already
    images = ax.get_images()
    if len(images) == 0:
        # TODO catch in add_scalebar -> provide better error message including that we
        # can just pass in color explicitly
        raise RuntimeError('has imshow() [/ similar] been called on this Axes yet?')

    assert len(images) == 1
    im = images[0]
    # TODO just get directly from background?
    arr = im.get_array()

    # TODO delete
    # for 192x192 frames, this seems to be about the section where depth title +
    # scalebar are currently drawn.
    #lowish_img_value = arr[:40, :].mean()

    # get warning "'partition' will ignore the 'mask' of the MaskedArray." if we don't
    # convert from MaskedArray w/ np.array(arr) first
    lowish_img_value = np.percentile(np.array(arr).flatten(), 15)

    # should be subclass of Normalize if not str
    assert type(im.norm) is not str
    # should map image value to [0.0, 1.0] (which should be input range of cmap)
    cmap_input = im.norm(lowish_img_value)
    lowish_value_color = im.cmap(cmap_input)

    # excluding alpha channel (which should be 1.0 anyway)
    luminance = np.mean(lowish_value_color[:3])

    if luminance > 0.5:
        # black
        default_color_over_image = 'k'
    else:
        default_color_over_image = 'w'

    return default_color_over_image

def set_facetgrid_legend(facet_grid, **kwargs) -> None:
    """Adds legend to FacetGrid, with hue artists from all its axes.

    In cases where different axes have different subsets of the hue levels,
    the legend may not contain the artists for the union of hue levels across
    all axes. This sets a legend from the hue artists across all axes.

    `kwargs` are passed to `facet_grid.add_legend`.
    """
    #from matplotlib.collections import PathCollection
    legend_data = dict()
    for ax in facet_grid.axes.flat:
        handles, labels = ax.get_legend_handles_labels()
        for label, h in zip(labels, handles):
            #if type(h) is PathCollection:
            # From inspecting facet_grid._legend_data in cases where some labels
            # pointed to empty lines (the phenotype in the case where things
            # weren't behaving as I wanted), the empty lines had this empty
            # facecolor.
            facecolor = h.get_facecolor()
            if len(facecolor) == 0:
                continue
            #else:
            #    print(type(h))
            #    import ipdb; ipdb.set_trace()

            if label in legend_data:
                # TODO maybe assert a wide variety of properties of the
                # matplotlib.collections.PathCollection objects are the same
                # (line width, dash, etc)
                past_facecolor = legend_data[label].get_facecolor()
                # TODO TODO TODO fix! this is failing again 2020-08-25

                # kc_mix_analysis.py w/ no just -j arg)
                assert np.array_equal(facecolor, past_facecolor), \
                    f'{facecolor} != {past_facecolor}'
            else:
                legend_data[label] = h

    facet_grid.add_legend(legend_data, **kwargs)

def showsync(thorsync_dir, verbose=False, **kwargs):

    Args:

    # TODO TODO make it possible to click to toggle display of lines like in thorsync
    # itself

    app = pg.mkQApp()

    before = time.time()
    print('loading HDF5...', flush=True, end='\n' if verbose else '')

    df = thor.load_thorsync_hdf5(thorsync_dir, verbose=verbose, **kwargs)

    took_s = time.time() - before
    print(f'done ({took_s:.1f}s)')

    # TODO set title to last 3 parts of path if parent two directories can be parsed as
    # (date, fly_num) (or if just under raw_data_root maybe?)
    title = str(thorsync_dir)

    # plot_widget if of type pyqtgraph.widgets.PlotWidget.PlotWidget
    # not really sure how 'all' and 'pairs' differ... ('pairs' skip some?)
    # TODO wanted to set clipToView=True and some other stuff here, but downsampling=100
    # didn't seem to work at all (unlike call below), so i'm not sure anythere here
    # will.
    plot_widget = pg.plot(title=title)

    plot_widget.setLabels(bottom='Seconds')

    # Passing `autoDownsample=True` to either the `plot` call above or those in the loop
    # below caused the startup time to increase beyond what is acceptable. Was also slow
    # after loading. Not sure why it only seems to work here.
    #plot_widget.setDownsampling(100)
    # TODO TODO could downsampling cause problems? how? maybe add cli flag to disable
    # it?
    # TODO TODO in `auto` case, is automatically selected factor changed when zooming?
    # (or would i (could i even?) need to link settting it to change in scale or
    # something?)
    plot_widget.setDownsampling(auto=True)

    # Much snappier with this. Doesn't try to draw data out of bounds of view box.
    plot_widget.setClipToView(True)

    # TODO set initial view box and / or build in margin such that legend is out of way
    # of traces immediately

    plot_widget.addLegend()

    time_col = thor.time_col

    x_min = 0
    x_max = df[time_col].iat[-1]
    y_min = -1
    y_max = 6
    # TODO maybe add a margin around this, but only if it can be made clear where the
    # region the data actually exists in is (not just black everywhere...)
    plot_widget.setLimits(xMin=x_min, xMax=x_max, yMin=y_min, yMax=y_max)

    plot_cols = [c for c in df.columns if c != time_col]
    for i, c in enumerate(plot_cols):
        if c == 'frame_counter':
            if verbose:
                print('not plotting frame_counter')
            continue

        col = df[c]
        # Otherwise this produces an in error in pyqtgraph internals

            col = col.astype(np.dtype('float64'))

        # TODO maybe store hardcoded map for colors for these?
        # not sure how consistent column order is as loaded from hdf5...
        plot_widget.plot(df[time_col], col, name=c, pen=(i, len(plot_cols)))

    app.exec_()
